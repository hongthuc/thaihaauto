        function handleDriveImageError(imgElement, fallbackLabel) {
            const fileId = imgElement.getAttribute('data-id');
            const step = parseInt(imgElement.getAttribute('data-retry') || '0', 10);

            if (fileId && step === 0) {
                imgElement.setAttribute('data-retry', '1');
                imgElement.src = 'https://lh3.googleusercontent.com/d/' + fileId + '=w1000';
            } else if (fileId && step === 1) {
                imgElement.setAttribute('data-retry', '2');
                imgElement.src = 'https://drive.google.com/uc?export=view&id=' + fileId;
            } else {
                imgElement.onerror = null;

        let currentBannerIndex = 0;
        const totalBannerSlides = 2;
        let bannerInterval = null;

        function updateBannerSlides(index) {
            currentBannerIndex = (index + totalBannerSlides) % totalBannerSlides;

            const desktopSlides = document.querySelectorAll('.desktop-slide');
            const mobileSlides = document.querySelectorAll('.mobile-slide');

            desktopSlides.forEach((slide, idx) => {
                if (idx === currentBannerIndex) {
                    slide.classList.remove('opacity-0', 'invisible');
                    slide.classList.add('opacity-100', 'visible');
                } else {
                    slide.classList.remove('opacity-100', 'visible');
                    slide.classList.add('opacity-0', 'invisible');
                }
            });

            mobileSlides.forEach((slide, idx) => {
                if (idx === currentBannerIndex) {
                    slide.classList.remove('opacity-0', 'invisible');
                    slide.classList.add('opacity-100', 'visible');
                } else {
                    slide.classList.remove('opacity-100', 'visible');
                    slide.classList.add('opacity-0', 'invisible');
                }
            });

            for (let i = 0; i < totalBannerSlides; i++) {

                        dot.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
                    } else {
                        dot.className = 'w-2 h-2 rounded-full bg-white/60 transition-all';
                    }
                }
            }
        }

            resetBannerTimer();
